# ===== ADINET static site: ADD these lines BELOW the existing cPanel lines in public_html/.htaccess =====
# (Do not delete or replace what is already there; cPanel keeps its PHP/handler settings in that file.)

# Friendly 404 page (the site ships its own 404.html)
ErrorDocument 404 /404.html

# The share-preview images have no file extension, so tell Apache they are PNGs
<FilesMatch "^(opengraph|twitter)-image$">
  ForceType image/png
</FilesMatch>

# Faster loading (each block is skipped automatically if the module is not available)
<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/css application/javascript text/plain application/xml image/svg+xml
</IfModule>
<IfModule mod_headers.c>
  # files under /_next/static have content-hashed names, so they can be cached for a year
  <FilesMatch "\.(js|css|woff2)$">
    Header set Cache-Control "public, max-age=31536000, immutable"
  </FilesMatch>
</IfModule>
